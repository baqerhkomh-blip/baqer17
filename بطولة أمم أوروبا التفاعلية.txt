const ADMIN_SECRET = "2026";

export default function App() {
  // Inject Cairo Arabic font
  useEffect(() => {

    link.href = 'https://fonts.googleapis.com/css2?family=Cairo:wght@400;600;700;800;900&display=swap';

    document.head.appendChild(link);
  }, []);

  // Auth & Cloud State
  const [user, setUser] = useState(null);
  const [syncStatus, setSyncStatus] = useState('connecting'); // 'connecting', 'synced', 'error'

  // Admin Mode State
  const [isAdmin, setIsAdmin] = useState(false);
  const [showAdminModal, setShowAdminModal] = useState(false);
  const [passcode, setPasscode] = useState('');
  const [adminError, setAdminError] = useState('');

  // Navigation & Filters State
  const [activeTab, setActiveTab] = useState('groups'); // 'groups', 'matches', 'thirds', 'knockout'
  const [selectedRound, setSelectedRound] = useState(1); // 1, 2, 3, or 'all'
  const [selectedGroupFilter, setSelectedGroupFilter] = useState('ALL');
  const [confirmModal, setConfirmModal] = useState(null);

  // Tournament Scores State (Synced with Firestore)
  const [groupScores, setGroupScores] = useState({});
  const [knockoutScores, setKnockoutScores] = useState({});

  // 1. Firebase Auth Effect
  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (err) {
        console.error('Auth error:', err);
        setSyncStatus('error');
      }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, (currentUser) => {
      setUser(currentUser);
    });
    return () => unsubscribe();
  }, []);

  // 2. Firestore Real-time Listener Effect
  useEffect(() => {
    if (!user) return;
    const stateDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'tournament', 'euro_state');
    const unsubscribe = onSnapshot(
      stateDocRef,
      (snapshot) => {
        if (snapshot.exists()) {
          const data = snapshot.data();
          setGroupScores(data.groupScores || {});
          setKnockoutScores(data.knockoutScores || {});
        }
        setSyncStatus('synced');
      },
      (err) => {
        console.error('Firestore sync error:', err);
        setSyncStatus('error');
      }
    );
    return () => unsubscribe();
  }, [user]);

  // Save updates to Firestore
  const saveTournamentToCloud = async (newGroupScores, newKnockoutScores) => {
    if (!user) return;
    try {
      const stateDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'tournament', 'euro_state');
      await setDoc(
        stateDocRef,
        {
          groupScores: newGroupScores,
          knockoutScores: newKnockoutScores,
          updatedAt: Date.now(),
          updatedBy: user.uid,
        },
        { merge: true }
      );
    } catch (err) {
      console.error('Failed to save to cloud:', err);
    }
  };

  // Admin Login Handler
  const handleAdminLogin = (e) => {
    e.preventDefault();
    if (passcode.trim() === ADMIN_SECRET) {
      setIsAdmin(true);
      setShowAdminModal(false);
      setPasscode('');
      setAdminError('');
    } else {
      setAdminError('الرمز السري غير صحيح! حاول مرة أخرى.');
    }
  };

  // Helper to get team by ID
  const teamsMap = useMemo(() => {
    const map = {};
    INITIAL_TEAMS.forEach((t) => { map[t.id] = t; });
    return map;
  }, []);

  // Calculate Group Standings
  const groupStandings = useMemo(() => {
    const standings = {};
    GROUPS.forEach((g) => {
      const groupTeams = INITIAL_TEAMS.filter((t) => t.group === g).map((t) => ({
        ...t,
        played: 0,
        won: 0,
        drawn: 0,
        lost: 0,
        gf: 0,
        ga: 0,
        gd: 0,
        pts: 0,
      }));

      const teamStatsMap = {};
      groupTeams.forEach((t) => { teamStatsMap[t.id] = t; });

      const gMatches = INITIAL_MATCHES.filter((m) => m.group === g);
      gMatches.forEach((m) => {
        const score = groupScores[m.id];
        if (score && score.home !== '' && score.away !== '' && score.home !== undefined && score.away !== undefined) {
          const hGoals = parseInt(score.home, 10);
          const aGoals = parseInt(score.away, 10);
          if (!isNaN(hGoals) && !isNaN(aGoals)) {
            const home = teamStatsMap[m.homeId];
            const away = teamStatsMap[m.awayId];
            home.played += 1;
            away.played += 1;
            home.gf += hGoals;
            home.ga += aGoals;
            home.gd = home.gf - home.ga;
            away.gf += aGoals;
            away.ga += hGoals;
            away.gd = away.gf - away.ga;

            if (hGoals > aGoals) {
              home.won += 1;
              home.pts += 3;
              away.lost += 1;
            } else if (hGoals < aGoals) {
              away.won += 1;
              away.pts += 3;
              home.lost += 1;
            } else {
              home.drawn += 1;
              away.drawn += 1;
              home.pts += 1;
              away.pts += 1;
            }
          }
        }
      });

      // Sort by Points -> GD -> GF -> Initial Position
      groupTeams.sort((a, b) => {
        if (b.pts !== a.pts) return b.pts - a.pts;
        if (b.gd !== a.gd) return b.gd - a.gd;
        if (b.gf !== a.gf) return b.gf - a.gf;
        return a.pos - b.pos;
      });

      standings[g] = groupTeams;
    });
    return standings;
  }, [groupScores]);

  // Calculate Best 3rd-Placed Teams
  const thirdPlacedTeams = useMemo(() => {
    const thirds = GROUPS.map((g) => groupStandings[g][2]);
    return [...thirds].sort((a, b) => {
      if (b.pts !== a.pts) return b.pts - a.pts;
      if (b.gd !== a.gd) return b.gd - a.gd;
      if (b.gf !== a.gf) return b.gf - a.gf;
      if (b.won !== a.won) return b.won - a.won;
      return a.group.localeCompare(b.group);
    });
  }, [groupStandings]);

  const qualifyingThirdIds = useMemo(() => {
    return new Set(thirdPlacedTeams.slice(0, 4).map((t) => t.id));
  }, [thirdPlacedTeams]);

  // Build Knockout Bracket Structure dynamically
  const knockoutBracket = useMemo(() => {
    const top4Thirds = thirdPlacedTeams.slice(0, 4);
    const key = top4Thirds.map((t) => t.group).sort().join('');
    const mapping = EURO_THIRD_PLACE_TABLE[key] || EURO_THIRD_PLACE_TABLE['ABCD'];

    const getThirdTeamByGroup = (grp) => groupStandings[grp][2];

    // Round of 16 Matchups (Official Euro Format)
    const r16Matches = [
      {
        id: 'R16_1',
        title: 'مباراة 1',
        labelHome: 'أول المجموعة B',
        labelAway: `ثالث المجموعة ${mapping['1B']}`,
        home: groupStandings['B'][0],
        away: getThirdTeamByGroup(mapping['1B']),
      },
      {
        id: 'R16_2',
        title: 'مباراة 2',
        labelHome: 'أول المجموعة A',
        labelAway: 'ثاني المجموعة C',
        home: groupStandings['A'][0],
        away: groupStandings['C'][1],
      },
      {
        id: 'R16_3',
        title: 'مباراة 3',
        labelHome: 'أول المجموعة F',
        labelAway: `ثالث المجموعة ${mapping['1F']}`,
        home: groupStandings['F'][0],
        away: getThirdTeamByGroup(mapping['1F']),
      },
      {
        id: 'R16_4',
        title: 'مباراة 4',
        labelHome: 'ثاني المجموعة D',
        labelAway: 'ثاني المجموعة E',
        home: groupStandings['D'][1],
        away: groupStandings['E'][1],
      },
      {
        id: 'R16_5',
        title: 'مباراة 5',
        labelHome: 'أول المجموعة E',
        labelAway: `ثالث المجموعة ${mapping['1E']}`,
        home: groupStandings['E'][0],
        away: getThirdTeamByGroup(mapping['1E']),
      },
      {
        id: 'R16_6',
        title: 'مباراة 6',
        labelHome: 'أول المجموعة D',
        labelAway: 'ثاني المجموعة F',
        home: groupStandings['D'][0],
        away: groupStandings['F'][1],
      },
      {
        id: 'R16_7',
        title: 'مباراة 7',
        labelHome: 'أول المجموعة C',
        labelAway: `ثالث المجموعة ${mapping['1C']}`,
        home: groupStandings['C'][0],
        away: getThirdTeamByGroup(mapping['1C']),
      },
      {
        id: 'R16_8',
        title: 'مباراة 8',
        labelHome: 'ثاني المجموعة A',
        labelAway: 'ثاني المجموعة B',
        home: groupStandings['A'][1],
        away: groupStandings['B'][1],
      },
    ];

    // Helper to resolve winner of a knockout match
    const getWinner = (match) => {
      if (!match.home || !match.away) return null;
      const sc = knockoutScores[match.id];
      if (!sc) return null;

      const h = sc.home !== '' && sc.home !== undefined ? parseInt(sc.home, 10) : NaN;
      const a = sc.away !== '' && sc.away !== undefined ? parseInt(sc.away, 10) : NaN;

      if (!isNaN(h) && !isNaN(a)) {
        if (h > a) return match.home;
        if (a > h) return match.away;
        if (sc.winnerId === match.home.id) return match.home;
        if (sc.winnerId === match.away.id) return match.away;
      }
      if (sc.winnerId === match.home.id) return match.home;
      if (sc.winnerId === match.away.id) return match.away;

      return null;
    };

    const r16WithWinners = r16Matches.map((m) => ({ ...m, winner: getWinner(m) }));

    // Quarter-finals
    const qfMatches = [
      {
        id: 'QF_1',
        title: 'ربع النهائي 1',
        labelHome: 'الفائز من مباراة 1',
        labelAway: 'الفائز من مباراة 2',
        home: r16WithWinners[0].winner,
        away: r16WithWinners[1].winner,
      },
      {
        id: 'QF_2',
        title: 'ربع النهائي 2',
        labelHome: 'الفائز من مباراة 3',
        labelAway: 'الفائز من مباراة 4',
        home: r16WithWinners[2].winner,
        away: r16WithWinners[3].winner,
      },
      {
        id: 'QF_3',
        title: 'ربع النهائي 3',
        labelHome: 'الفائز من مباراة 5',
        labelAway: 'الفائز من مباراة 6',
        home: r16WithWinners[4].winner,
        away: r16WithWinners[5].winner,
      },
      {
        id: 'QF_4',
        title: 'ربع النهائي 4',
        labelHome: 'الفائز من مباراة 7',
        labelAway: 'الفائز من مباراة 8',
        home: r16WithWinners[6].winner,
        away: r16WithWinners[7].winner,
      },
    ];
    const qfWithWinners = qfMatches.map((m) => ({ ...m, winner: getWinner(m) }));

    // Semi-finals
    const sfMatches = [
      {
        id: 'SF_1',
        title: 'نصف النهائي 1',
        labelHome: 'الفائز من ربع النهائي 1',
        labelAway: 'الفائز من ربع النهائي 2',
        home: qfWithWinners[0].winner,
        away: qfWithWinners[1].winner,
      },
      {
        id: 'SF_2',
        title: 'نصف النهائي 2',
        labelHome: 'الفائز من ربع النهائي 3',
        labelAway: 'الفائز من ربع النهائي 4',
        home: qfWithWinners[2].winner,
        away: qfWithWinners[3].winner,
      },
    ];
    const sfWithWinners = sfMatches.map((m) => ({ ...m, winner: getWinner(m) }));

    // Final
    const finalMatch = {
      id: 'FINAL',
      title: 'نهائي اليورو 🏆',
      labelHome: 'الفائز من نصف النهائي 1',
      labelAway: 'الفائز من نصف النهائي 2',
      home: sfWithWinners[0].winner,
      away: sfWithWinners[1].winner,
    };
    const finalWithWinner = { ...finalMatch, winner: getWinner(finalMatch) };

    return {
      r16: r16WithWinners,
      qf: qfWithWinners,
      sf: sfWithWinners,
      final: finalWithWinner,
    };
  }, [groupStandings, thirdPlacedTeams, knockoutScores]);

  // Handlers for updating scores (Admin only)
  const handleGroupScoreChange = (matchId, side, val) => {
    if (!isAdmin) return;
    if (val !== '' && (isNaN(val) || parseInt(val, 10) < 0 || parseInt(val, 10) > 99)) return;
    const updated = {
      ...groupScores,
      [matchId]: {
        ...(groupScores[matchId] || { home: '', away: '' }),
        [side]: val,
      },
    };
    setGroupScores(updated);
    saveTournamentToCloud(updated, knockoutScores);
  };

  const handleKnockoutScoreChange = (matchId, side, val) => {
    if (!isAdmin) return;
    if (val !== '' && (isNaN(val) || parseInt(val, 10) < 0 || parseInt(val, 10) > 99)) return;
    const updated = {
      ...knockoutScores,
      [matchId]: {
        ...(knockoutScores[matchId] || { home: '', away: '', winnerId: null }),
        [side]: val,
      },
    };
    setKnockoutScores(updated);
    saveTournamentToCloud(groupScores, updated);
  };

  const handleSelectKnockoutWinner = (matchId, teamId) => {
    if (!isAdmin) return;
    const updated = {
      ...knockoutScores,
      [matchId]: {
        ...(knockoutScores[matchId] || { home: '', away: '' }),
        winnerId: knockoutScores[matchId]?.winnerId === teamId ? null : teamId,
      },
    };
    setKnockoutScores(updated);
    saveTournamentToCloud(groupScores, updated);
  };

  // Simulate Random Group Scores (Admin Only)
  const simulateGroupMatches = () => {
    if (!isAdmin) return;
    const randomScores = {};
    INITIAL_MATCHES.forEach((m) => {
      randomScores[m.id] = {
        home: String(Math.floor(Math.random() * 4)),
        away: String(Math.floor(Math.random() * 4)),
      };
    });
    setGroupScores(randomScores);
    saveTournamentToCloud(randomScores, knockoutScores);
  };

  // Reset all scores (Admin Only)
  const resetAllScores = () => {
    if (!isAdmin) return;
    setGroupScores({});
    setKnockoutScores({});
    saveTournamentToCloud({}, {});
    setConfirmModal(null);
  };

  // Count completed group matches
  const completedMatchesCount = useMemo(() => {
    return INITIAL_MATCHES.filter((m) => {
      const s = groupScores[m.id];
      return s && s.home !== '' && s.away !== '' && s.home !== undefined && s.away !== undefined;
    }).length;
  }, [groupScores]);

  return (
    <div 
      dir="rtl"

                                    value={sc.home}
                                    onChange={(e) => handleGroupScoreChange(m.id, 'home', e.target.value)}
                                    className="w-11 h-11 sm:w-12 sm:h-12 rounded-xl bg-[#00262d] border-2 border-[#e5a93c]/70 focus:border-emerald-400 focus:outline-none text-center text-lg font-black text-white shadow-inner"
                                  />
                                  <span className="text-cyan-300 font-black text-xs">:</span>
                                  <input
                                    type="number"
                                    min="0"
                                    max="99"
                                    placeholder="-"
                                    value={sc.away}
                                    onChange={(e) => handleGroupScoreChange(m.id, 'away', e.target.value)}
                                    className="w-11 h-11 sm:w-12 sm:h-12 rounded-xl bg-[#00262d] border-2 border-[#e5a93c]/70 focus:border-emerald-400 focus:outline-none text-center text-lg font-black text-white shadow-inner"
                                  />
                                </div>
                              ) : (
                                <div className="shrink-0">
                                  {isFinished ? (
                                    <div className="px-3.5 py-2 rounded-xl bg-[#00262d] border border-emerald-400/40 flex items-center gap-2 font-black text-lg text-white shadow-inner">
                                      <span className={hScore > aScore ? 'text-[#e5a93c]' : ''}>{hScore}</span>
                                      <span className="text-cyan-400 text-xs">-</span>
                                      <span className={aScore > hScore ? 'text-[#e5a93c]' : ''}>{aScore}</span>
                                    </div>
                                  ) : (
                                    <div className="px-3.5 py-2 rounded-xl bg-[#002f38]/80 border border-cyan-400/20 font-black text-xs text-cyan-300">
                                      VS
                                    </div>
                                  )}
                                </div>
                              )}

                              {/* Away Team */}
                              <div className="flex-1 flex items-center justify-end gap-2.5 text-left min-w-0">
                                <div className="truncate text-left">
                                  <div
                                    className={`font-black text-sm sm:text-base truncate ${
                                      isFinished && aScore > hScore ? 'text-[#e5a93c]' : 'text-white'
                                    }`}
                                  >
                                    {away.player}
                                  </div>
                                  <div className="text-xs text-cyan-200 font-semibold">{away.country}</div>
                                </div>
                                <span className="w-10 h-10 rounded-full bg-[#002f38] border border-white/20 flex items-center justify-center text-xl shrink-0 shadow">
                                  {away.flag}
                                </span>
                              </div>
                            </div>
                          </div>
                        );
                      })}
                    </div>
                  </div>
                );
              })}
          </div>
        )}

        {/* ==================== TAB 3: BEST 3RD-PLACED TEAMS ==================== */}
        {activeTab === 'thirds' && (
          <div className="space-y-6">
            <div className="bg-[#004956] border-2 border-[#e5a93c]/70 rounded-2xl overflow-hidden shadow-xl">
              <div className="bg-[#00343e] px-6 py-4 border-b border-cyan-400/30 flex flex-col sm:flex-row sm:items-center justify-between gap-2">
                <div>
                  <h2 className="text-xl font-black text-white flex items-center gap-2">
                    <Award className="w-6 h-6 text-[#e5a93c]" />
                    <span>جدول ترتيب أفضل المنتخبات الحاصلة على المركز الثالث</span>
                  </h2>
                  <p className="text-xs sm:text-sm text-cyan-200 mt-1">
                    يتأهل أفضل 4 منتخبات من أصل 6 مجموعات إلى دور الـ 16 بناءً على (النقاط، ثم فارق الأهداف، ثم الأهداف المسجلة)
                  </p>
                </div>
              </div>

              <div className="overflow-x-auto">
                <table className="w-full text-right border-collapse">
                  <thead>
                    <tr className="bg-[#003d48] text-cyan-200 text-xs sm:text-sm font-bold border-b border-cyan-400/20">
                      <th className="py-3.5 px-4 text-center">الترتيب</th>
                      <th className="py-3.5 px-3 text-center">المجموعة</th>
                      <th className="py-3.5 px-4">المشارك والمنتخب</th>
                      <th className="py-3.5 px-3 text-center">لعب</th>
                      <th className="py-3.5 px-3 text-center">فوز</th>
                      <th className="py-3.5 px-3 text-center">تعادل</th>
                      <th className="py-3.5 px-3 text-center">خسارة</th>
                      <th className="py-3.5 px-3 text-center">له</th>
                      <th className="py-3.5 px-3 text-center">عليه</th>
                      <th className="py-3.5 px-3 text-center">الفارق</th>
                      <th className="py-3.5 px-4 text-center text-[#e5a93c]">النقاط</th>
                      <th className="py-3.5 px-4 text-center">الحالة</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-cyan-400/20 text-sm">
                    {thirdPlacedTeams.map((team, index) => {
                      const isQualified = index < 4;
                      return (
                        <tr
                          key={team.id}
                          className={`transition ${
                            isQualified ? 'bg-emerald-950/25 hover:bg-emerald-900/30' : 'bg-rose-950/20 opacity-75'
                          }`}
                        >
                          <td className="py-4 px-4 text-center font-black">
                            <span
                              className={`inline-flex items-center justify-center w-7 h-7 rounded-full text-xs ${
                                isQualified ? 'bg-emerald-400 text-slate-950' : 'bg-rose-500/40 text-white'
                              }`}
                            >
                              {index + 1}
                            </span>
                          </td>
                          <td className="py-4 px-3 text-center">
                            <span className="px-2.5 py-1 rounded-lg bg-[#002d36] font-black text-[#e5a93c] border border-[#e5a93c]/30">
                              {team.group}
                            </span>
                          </td>
                          <td className="py-4 px-4">
                            <div className="flex items-center gap-3">
                              <span className="w-9 h-9 rounded-full bg-[#002d36] border border-white/20 flex items-center justify-center text-xl">
                                {team.flag}
                              </span>
                              <div>
                                <div className="font-black text-white text-base">{team.player}</div>
                                <div className="text-xs text-cyan-200">{team.country}</div>
                              </div>
                            </div>
                          </td>
                          <td className="py-4 px-3 text-center font-bold">{team.played}</td>
                          <td className="py-4 px-3 text-center text-emerald-300 font-bold">{team.won}</td>
                          <td className="py-4 px-3 text-center text-amber-200 font-bold">{team.drawn}</td>
                          <td className="py-4 px-3 text-center text-rose-300 font-bold">{team.lost}</td>
                          <td className="py-4 px-3 text-center font-mono">{team.gf}</td>
                          <td className="py-4 px-3 text-center font-mono">{team.ga}</td>
                          <td className="py-4 px-3 text-center font-mono font-bold">
                            {team.gd > 0 ? `+${team.gd}` : team.gd}
                          </td>
                          <td className="py-4 px-4 text-center">
                            <span className="px-3 py-1 rounded-lg bg-[#00262d] text-[#e5a93c] font-black text-base border border-[#e5a93c]/40">
                              {team.pts}
                            </span>
                          </td>
                          <td className="py-4 px-4 text-center">
                            {isQualified ? (
                              <span className="inline-flex items-center gap-1 px-3 py-1 rounded-full text-xs font-extrabold bg-emerald-500/20 text-emerald-300 border border-emerald-400/30">
                                متأهل لدور الـ 16
                              </span>
                            ) : (
                              <span className="inline-flex items-center gap-1 px-3 py-1 rounded-full text-xs font-bold bg-rose-500/20 text-rose-300 border border-rose-400/30">
                                خارج التأهل
                              </span>
                            )}
                          </td>
                        </tr>
                      );
                    })}
                  </tbody>
                </table>
              </div>
            </div>
          </div>
        )}

        {/* ==================== TAB 4: KNOCKOUT BRACKET ==================== */}
        {activeTab === 'knockout' && (
          <div className="space-y-8">
            <div className="bg-[#00424d] border border-cyan-400/30 rounded-xl p-4 flex items-center gap-3 text-xs sm:text-sm text-cyan-100">
              <Info className="w-5 h-5 text-[#e5a93c] shrink-0" />
              {isAdmin ? (
                <p>
                  <strong>وضع المشرف مفعل:</strong> يمكنك إدخال نتائج الأشواط الأصلية أو الضغط مباشرة على اسم المشارك لتحديد المتأهل (في حال الحسم بركلات الترجيح).
                </p>
              ) : (
                <p>
                  يتم توزيع الفرق المتأهلة إلى <strong>دور الـ 16</strong> تلقائياً بناءً على ترتيب المجموعات وأفضل الثوالث. يقوم مشرف البطولة بتحديث نتائج الأدوار الإقصائية.
                </p>
              )}
            </div>

            {/* Round of 16 */}
            <div>
              <h3 className="text-xl font-black text-[#e5a93c] mb-4 flex items-center gap-2">
                <Shield className="w-5 h-5" />
                <span>مواجهات دور الـ 16</span>
              </h3>
              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
                {knockoutBracket.r16.map((m) => (
                  <KnockoutMatchCard
                    key={m.id}
                    match={m}
                    isAdmin={isAdmin}
                    scoreData={knockoutScores[m.id]}
                    onScoreChange={handleKnockoutScoreChange}
                    onSelectWinner={handleSelectKnockoutWinner}
                  />
                ))}
              </div>
            </div>

            {/* Quarter-Finals */}
            <div>
              <h3 className="text-xl font-black text-[#e5a93c] mb-4 flex items-center gap-2">
                <Flame className="w-5 h-5" />
                <span>الدور ربع النهائي (دور الـ 8)</span>
              </h3>
              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
                {knockoutBracket.qf.map((m) => (
                  <KnockoutMatchCard
                    key={m.id}
                    match={m}
                    isAdmin={isAdmin}
                    scoreData={knockoutScores[m.id]}
                    onScoreChange={handleKnockoutScoreChange}
                    onSelectWinner={handleSelectKnockoutWinner}
                  />
                ))}
              </div>
            </div>

            {/* Semi-Finals & Final */}
            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6 items-start">
              <div className="lg:col-span-2 space-y-4">
                <h3 className="text-xl font-black text-[#e5a93c] flex items-center gap-2">
                  <Sparkles className="w-5 h-5" />
                  <span>الدور نصف النهائي</span>
                </h3>
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  {knockoutBracket.sf.map((m) => (
                    <KnockoutMatchCard
                      key={m.id}
                      match={m}
                      isAdmin={isAdmin}
                      scoreData={knockoutScores[m.id]}
                      onScoreChange={handleKnockoutScoreChange}
                      onSelectWinner={handleSelectKnockoutWinner}
                    />
                  ))}
                </div>
              </div>

              <div className="space-y-4">
                <h3 className="text-xl font-black text-[#e5a93c] flex items-center gap-2">
                  <Trophy className="w-5 h-5" />
                  <span>المباراة النهائية</span>
                </h3>
                <KnockoutMatchCard
                  match={knockoutBracket.final}
                  isAdmin={isAdmin}
                  scoreData={knockoutScores[knockoutBracket.final.id]}
                  onScoreChange={handleKnockoutScoreChange}
                  onSelectWinner={handleSelectKnockoutWinner}
                  isFinal
                />
              </div>
            </div>
          </div>
        )}
      </main>

      {/* ==================== ADMIN LOGIN MODAL ==================== */}
      {showAdminModal && (
        <div className="fixed inset-0 z-50 bg-black/75 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-[#003640] border-2 border-[#e5a93c] rounded-2xl max-w-sm w-full p-6 text-right shadow-2xl">
            <div className="flex items-center justify-center w-14 h-14 rounded-2xl bg-[#e5a93c]/20 border border-[#e5a93c]/40 text-[#e5a93c] mx-auto mb-4">
              <KeyRound className="w-7 h-7" />
            </div>
            <h3 className="text-xl font-black text-white text-center mb-1">
              🔐 بوابة مشرف البطولة
            </h3>
            <p className="text-cyan-200/80 text-xs text-center mb-5">
              أدخل الرمز السري لتفعيل صلاحية تعديل نتائج المباريات
            </p>

            <form onSubmit={handleAdminLogin} className="space-y-4">
              <div>
                <input
                  type="password"
                  placeholder="أدخل الرمز السري..."
                  value={passcode}
                  onChange={(e) => {
                    setPasscode(e.target.value);
                    setAdminError('');
                  }}
                  autoFocus
                  className="w-full px-4 py-3 rounded-xl bg-[#002229] text-white border-2 border-cyan-400/30 focus:border-[#e5a93c] focus:outline-none text-center text-lg font-black tracking-widest"
                />
                {adminError && (
                  <div className="flex items-center justify-center gap-1.5 text-rose-400 text-xs font-bold mt-2">
                    <AlertCircle className="w-4 h-4" />
                    <span>{adminError}</span>
                  </div>
                )}
              </div>

              <div className="flex items-center gap-2.5">
                <button
                  type="submit"
                  className="flex-1 py-2.5 rounded-xl bg-gradient-to-l from-[#e5a93c] to-[#d4911e] text-[#00262d] font-black text-sm hover:opacity-95 transition cursor-pointer"
                >
                  تفعيل وضع الأدمن
                </button>
                <button
                  type="button"
                  onClick={() => {
                    setShowAdminModal(false);
                    setAdminError('');
                    setPasscode('');
                  }}
                  className="flex-1 py-2.5 rounded-xl bg-[#00262d] text-cyan-200 font-bold text-sm hover:bg-[#002f38] transition cursor-pointer"
                >
                  إلغاء
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* Custom Confirmation Modal for Reset */}
      {confirmModal === 'reset' && (
        <div className="fixed inset-0 z-50 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-[#003b46] border-2 border-[#e5a93c] rounded-2xl max-w-md w-full p-6 text-right shadow-2xl">
            <h3 className="text-xl font-black text-white mb-2">تأكيد تصفير البطولة</h3>
            <p className="text-cyan-100 text-sm mb-6">
              هل أنت متأكد من مسح جميع نتائج مباريات المجموعات والأدوار الإقصائية والبدء من جديد؟ سيتم تحديث ذلك لدى جميع المشاركين.
            </p>
            <div className="flex items-center justify-end gap-3">
              <button
                onClick={() => setConfirmModal(null)}
                className="px-4 py-2 rounded-xl bg-[#00262d] text-cyan-200 font-bold text-sm hover:bg-[#00313a] cursor-pointer"
              >
                إلغاء
              </button>
              <button
                onClick={resetAllScores}
                className="px-4 py-2 rounded-xl bg-[#e61c5d] text-white font-bold text-sm hover:bg-rose-600 cursor-pointer"
              >
                نعم، تصفير الكل
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

// --- SUB-COMPONENT FOR KNOCKOUT MATCH CARD ---
function KnockoutMatchCard({ match, isAdmin, scoreData, onScoreChange, onSelectWinner, isFinal }) {
  const sc = scoreData || { home: '', away: '', winnerId: null };

  const renderTeamRow = (team, fallbackLabel, side) => {
    const isWinner = match.winner && team && match.winner.id === team.id;
    const hasScore = sc[side] !== '' && sc[side] !== undefined;

    return (
      <div
        className={`flex items-center justify-between gap-2 p-2.5 rounded-xl border transition ${
          isWinner
            ? 'bg-emerald-950/40 border-emerald-400/70'
            : 'bg-[#002f38]/90 border-cyan-400/15'
        }`}
      >
        {team ? (
          <div
            onClick={() => isAdmin && onSelectWinner(match.id, team.id)}
            className={`flex items-center gap-2.5 text-right flex-1 min-w-0 ${
              isAdmin ? 'cursor-pointer group' : ''
            }`}
            title={isAdmin ? 'اضغط لتحديد الفائز المتأهل' : ''}
          >
            <span className="w-8 h-8 rounded-full bg-[#002229] border border-white/20 flex items-center justify-center text-lg shrink-0">
              {team.flag}
            </span>
            <div className="truncate">
              <div
                className={`font-black text-sm truncate transition ${
                  isAdmin ? 'group-hover:text-[#e5a93c]' : ''
                } ${isWinner ? 'text-emerald-300' : 'text-white'}`}
              >
                {team.player}
              </div>
              <div className="text-[11px] text-cyan-200/80">
                {team.country} ({team.group})
              </div>
            </div>
          </div>
        ) : (
          <div className="flex items-center gap-2.5 text-right flex-1 py-1">
            <span className="w-8 h-8 rounded-full bg-[#002229] border border-dashed border-cyan-400/30 flex items-center justify-center text-xs text-cyan-400">
              ؟
            </span>
            <span className="text-xs text-cyan-300/60 font-semibold">{fallbackLabel}</span>
          </div>
        )}

        {isAdmin ? (
          <input
            type="number"
            min="0"
            max="99"
            disabled={!team}
            placeholder="-"
            value={sc[side]}
            onChange={(e) => onScoreChange(match.id, side, e.target.value)}
            className="w-10 h-10 rounded-lg bg-[#001f24] border border-[#e5a93c]/60 focus:border-emerald-400 focus:outline-none text-center font-black text-sm text-white disabled:opacity-40"
          />
        ) : (
          <div className="w-9 h-9 rounded-lg bg-[#001f24] border border-cyan-400/20 flex items-center justify-center font-black text-sm text-white">
            {hasScore ? sc[side] : '-'}
          </div>
        )}
      </div>
    );
  };

  return (
    <div
      className={`rounded-2xl p-4 shadow-xl border-2 flex flex-col justify-between gap-3 ${
        isFinal
          ? 'bg-gradient-to-b from-[#005563] to-[#00343d] border-[#e5a93c]'
          : 'bg-[#004854] border-cyan-400/30'
      }`}
    >
      <div className="flex items-center justify-between text-xs border-b border-cyan-300/20 pb-2">
        <span className="font-black text-[#e5a93c]">{match.title}</span>
        <span className="text-[11px] text-cyan-200/80">
          {match.winner
            ? `تأهل: ${match.winner.player}`
            : isAdmin
            ? 'أدخل النتيجة أو اختر المتأهل'
            : 'لم تُحسم بعد'}
        </span>
      </div>

      <div className="space-y-2">
        {renderTeamRow(match.home, match.labelHome, 'home')}
        {renderTeamRow(match.away, match.labelAway, 'away')}
      </div>
    </div>
  );
}
